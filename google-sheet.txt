function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu("Inventory System")
    .addItem("Setup System", "setupInventorySystem")
    .addItem("Add New Item", "addNewItemPrompt")
    .addToUi();
}

/* =====================================================
   MAIN INSTALL
===================================================== */

function setupInventorySystem() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheets = [
    "Inventory_Master",
    "Stock_In",
    "Stock_Out",
    "Stock_Summary",
    "Movement_Log",
    "Inventory_Analytics"
  ];

  sheets.forEach(name => {
    const sh = ss.getSheetByName(name);
    if (sh) ss.deleteSheet(sh);
  });

  SpreadsheetApp.flush();
  sheets.forEach(name => ss.insertSheet(name));
  SpreadsheetApp.flush();

  setupInventoryMaster();
  setupStockSheets();
  setupSummary();
  setupMovementLog();
  setupAnalytics();

  SpreadsheetApp.getUi().alert("Inventory System Installed Successfully");
}

/* =====================================================
   HELPER — apply consistent table style
===================================================== */

function styleSheet(sh, headerRange, headerColor, numCols, numDataRows) {
  // Header
  headerRange
    .setFontWeight("bold")
    .setFontSize(11)
    .setFontFamily("Arial")
    .setHorizontalAlignment("center")
    .setVerticalAlignment("middle")
    .setBackground(headerColor)
    .setFontColor("#FFFFFF");

  sh.setFrozenRows(1);
  sh.setRowHeight(1, 36);

  // Data area
  sh.getRange(2, 1, numDataRows, numCols)
    .setHorizontalAlignment("center")
    .setVerticalAlignment("middle")
    .setFontSize(11)
    .setFontFamily("Arial")
    .setBackground("#FFFFFF");

  // Borders on full table
  sh.getRange(1, 1, numDataRows + 1, numCols).setBorder(
    true, true, true, true, true, true,
    "#CCCCCC", SpreadsheetApp.BorderStyle.SOLID
  );
}

/* =====================================================
   INVENTORY MASTER  — muted blue
===================================================== */

function setupInventoryMaster() {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Inventory_Master");
  sh.clear();

  const header = sh.getRange("A1:E1");
  header.setValues([["Item ID", "Item Name", "Category", "Unit", "Min Stock"]]);
  styleSheet(sh, header, "#5C7FA3", 5, 999);
  sh.setColumnWidths(1, 5, 160);

  const unitOptions = ["No","Kg","Gram","Litre","ML","Meter","Pack","Box","%"];
  const rule = SpreadsheetApp.newDataValidation()
    .requireValueInList(unitOptions)
    .setAllowInvalid(false)
    .build();
  sh.getRange("D2:D1000").setDataValidation(rule);
}

/* =====================================================
   STOCK IN / OUT  — muted green / muted rose
===================================================== */

function setupStockSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const master = ss.getSheetByName("Inventory_Master");
  const itemRange = master.getRange("B2:B");

  const config = {
    "Stock_In":  { color: "#4D8B6F" },   // muted green
    "Stock_Out": { color: "#A05252" }    // muted rose-red
  };

  Object.entries(config).forEach(([name, cfg]) => {
    const sh = ss.getSheetByName(name);
    sh.clear();

    const qtyLabel = name === "Stock_In" ? "Qty In" : "Qty Out";
    const header = sh.getRange("A1:E1");
    header.setValues([["Date", "Item Name", qtyLabel, "Unit", "Name"]]);
    styleSheet(sh, header, cfg.color, 5, 999);
    sh.setColumnWidths(1, 5, 160);

    sh.getRange("A2:A1000").setNumberFormat("dd/mm/yyyy");

    const rule = SpreadsheetApp.newDataValidation()
      .requireValueInRange(itemRange, true)
      .setAllowInvalid(false)
      .build();
    sh.getRange("B2:B1000").setDataValidation(rule);

    sh.getRange("D2")
      .setFormula('=ARRAYFORMULA(IF(B2:B="","",IFERROR(VLOOKUP(B2:B,Inventory_Master!B:D,3,false),"")))')
  });
}

/* =====================================================
   STOCK SUMMARY  — muted amber
===================================================== */

function setupSummary() {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Stock_Summary");
  sh.clear();

  const header = sh.getRange("A1:F1");
  header.setValues([["Item Name","Total In","Total Out","Current Stock","Unit","Min Stock"]]);
  styleSheet(sh, header, "#00695C", 6, 999);
  sh.setColumnWidths(1, 6, 160);

  sh.getRange("A2").setFormula('=FILTER(Inventory_Master!B2:B,Inventory_Master!B2:B<>"")');
  sh.getRange("B2").setFormula('=ARRAYFORMULA(IF(A2:A="","",IFERROR(VLOOKUP(A2:A,QUERY(Stock_In!B:C,"select B,sum(C) where B is not null group by B label sum(C) \'\'",0),2,false),0)))');
  sh.getRange("C2").setFormula('=ARRAYFORMULA(IF(A2:A="","",IFERROR(VLOOKUP(A2:A,QUERY(Stock_Out!B:C,"select B,sum(C) where B is not null group by B label sum(C) \'\'",0),2,false),0)))');
  sh.getRange("D2").setFormula('=ARRAYFORMULA(IF(A2:A="","",B2:B-C2:C))');
  sh.getRange("E2").setFormula('=ARRAYFORMULA(IF(A2:A="","",IFERROR(VLOOKUP(A2:A,Inventory_Master!B:D,3,false),"")))');;
  sh.getRange("F2").setFormula('=ARRAYFORMULA(IF(A2:A="","",IFERROR(VLOOKUP(A2:A,Inventory_Master!B:E,4,false),0)))');

  // Low stock = red text + pink bg, healthy = green text
  const lowStock = SpreadsheetApp.newConditionalFormatRule()
    .whenFormulaSatisfied('=AND($A2<>"",$D2<=$F2)')
    .setBackground("#FDECEA")
    .setFontColor("#C0392B")
    .setRanges([sh.getRange("D2:D1000")])
    .build();

  const goodStock = SpreadsheetApp.newConditionalFormatRule()
    .whenFormulaSatisfied('=AND($A2<>"",$D2>$F2)')
    .setFontColor("#1B8A4E")
    .setRanges([sh.getRange("D2:D1000")])
    .build();

  sh.setConditionalFormatRules([lowStock, goodStock]);
}

/* =====================================================
   MOVEMENT LOG  — muted purple, minimal, IN/OUT coloured
===================================================== */

function setupMovementLog() {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Movement_Log");
  sh.clear();

  const header = sh.getRange("A1:F1");
  header.setValues([["Timestamp","Type","Item Name","Quantity","Unit","Name"]]);
  styleSheet(sh, header, "#7A6B8A", 6, 999);   // muted purple
  sh.setColumnWidths(1, 6, 175);

  sh.getRange("A2:A1000").setNumberFormat("dd/mm/yyyy hh:mm");

  // IN = green, OUT = red on the Type column
  const inRule = SpreadsheetApp.newConditionalFormatRule()
    .whenTextEqualTo("IN")
    .setBackground("#E8F5E9")
    .setFontColor("#1B8A4E")
    .setRanges([sh.getRange("B2:B1000")])
    .build();

  const outRule = SpreadsheetApp.newConditionalFormatRule()
    .whenTextEqualTo("OUT")
    .setBackground("#FDECEA")
    .setFontColor("#C0392B")
    .setRanges([sh.getRange("B2:B1000")])
    .build();

  sh.setConditionalFormatRules([inRule, outRule]);
}

/* =====================================================
   AUTO DATE + LOGGING
===================================================== */

function onEdit(e) {
  const sh   = e.source.getActiveSheet();
  const name = sh.getName();

  if (name !== "Stock_In" && name !== "Stock_Out") return;
  if (e.range.getRow() < 2) return;

  const row  = e.range.getRow();
  const col  = e.range.getColumn();
  const ui   = SpreadsheetApp.getUi();

  const dateCell   = sh.getRange(row,1);
  const item       = sh.getRange(row,2).getValue();
  const qty        = sh.getRange(row,3).getValue();
  const unit       = sh.getRange(row,4).getValue();
  const personName = sh.getRange(row,5).getValue();

  // Auto-fill date when item + qty are present
  if (!dateCell.getValue() && item && qty) {
    dateCell.setValue(new Date());
  }

  const log  = e.source.getSheetByName("Movement_Log");
  const type = name === "Stock_In" ? "IN" : "OUT";

  if (col === 3 && item && qty) {
    log.appendRow([new Date(), type, item, qty, unit, personName]);

  } else if (col === 5 && item && qty && personName) {
    const lastRow = log.getLastRow();
    if (lastRow >= 2) {
      const data = log.getRange(2, 1, lastRow - 1, 6).getValues();
      for (let i = data.length - 1; i >= 0; i--) {
        if (data[i][1] === type && data[i][2] === item && data[i][5] === "") {
          log.getRange(i + 2, 6).setValue(personName);
          break;
        }
      }
    }
  }
}

/* =====================================================
   ADD ITEM
===================================================== */

function addNewItemPrompt() {
  const ui = SpreadsheetApp.getUi();
  const response = ui.prompt("Add New Item", "Enter the item name:", ui.ButtonSet.OK_CANCEL);

  if (response.getSelectedButton() == ui.Button.OK) {
    const itemName = response.getResponseText().trim();
    if (!itemName) { ui.alert("Item name cannot be empty."); return; }

    const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Inventory_Master");
    const lastRow = sh.getLastRow() + 1;
    const itemId = "ITM" + Utilities.formatString("%03d", lastRow - 1);

    sh.getRange(lastRow,1).setValue(itemId);
    sh.getRange(lastRow,2).setValue(itemName);
    sh.getRange(lastRow,5).setValue(5);

    sh.getRange(lastRow, 1, 1, 5)
      .setHorizontalAlignment("center")
      .setVerticalAlignment("middle")
      .setFontSize(11)
      .setFontFamily("Arial")
      .setBackground("#FFFFFF");

    ui.alert('Item "' + itemName + '" added successfully!');
  }
}
